<script data-bird>
(()=>{'use strict';
  const reduced=window.matchMedia('(prefers-reduced-motion: reduce)');
  document.querySelectorAll('.bird-slide').forEach(slide=>{
    const button=slide.querySelector('.bird-button'),stage=slide.querySelector('.bird-stage'),status=slide.querySelector('.bird-status');
    if(!button||!stage||!status)return;
    function stop(message){stage.classList.remove('bird-is-flying');button.setAttribute('aria-pressed','false');status.textContent=message;}
    function sync(){if(document.body.classList.contains('motion-off'))stop('全局动效已关闭，小鸟保持静止。');else if(reduced.matches)stop('系统已启用减少动态效果，小鸟保持静止。');else if(button.getAttribute('aria-pressed')==='false')status.textContent='当前静止，点击按钮开始。';}
    button.addEventListener('click',()=>{if(document.body.classList.contains('motion-off')||reduced.matches){sync();return;}if(button.getAttribute('aria-pressed')==='true'){stop('小鸟已停止。');return;}stage.classList.add('bird-is-flying');button.setAttribute('aria-pressed','true');status.textContent='小鸟正在扇动翅膀，再点一次停止。';});
    new MutationObserver(sync).observe(document.body,{attributes:true,attributeFilter:['class']});
    if(reduced.addEventListener)reduced.addEventListener('change',sync);else if(reduced.addListener)reduced.addListener(sync);
    sync();
  });
})();
</script>
